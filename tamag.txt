  <script>
    (function () {
      const els = {
        overlay: document.getElementById("nameOverlay"),
        beginBtn: document.getElementById("beginBtn"),
        nameInput: document.getElementById("petNameInput"),
        petNameLabel: document.getElementById("petNameLabel"),
        moodText: document.getElementById("moodText"),
        statusText: document.getElementById("statusText"),
        statusChip: document.getElementById("statusChip"),
        petFace: document.getElementById("petFace"),
        petWrapper: document.getElementById("petWrapper"),
        ageDisplay: document.getElementById("ageDisplay"),
        overallText: document.getElementById("overallText"),
        hungerValue: document.getElementById("hungerValue"),
        happinessValue: document.getElementById("happinessValue"),
        energyValue: document.getElementById("energyValue"),
        loveValue: document.getElementById("loveValue"),
        hungerBar: document.getElementById("hungerBar"),
        happinessBar: document.getElementById("happinessBar"),
        energyBar: document.getElementById("energyBar"),
        loveBar: document.getElementById("loveBar"),
        btnFeed: document.getElementById("btnFeed"),
        btnPlay: document.getElementById("btnPlay"),
        btnSleep: document.getElementById("btnSleep"),
        btnCuddle: document.getElementById("btnCuddle")
      };

      let state = { name: "", stats: { hunger: 100, happiness: 100, energy: 100, love: 100 }, birthTime: Date.now(), lastUpdated: Date.now() };

      const clamp = (v, a, b) => Math.max(a, Math.min(b, v));

      els.beginBtn.addEventListener("click", () => {
        const val = els.nameInput.value.trim();
        state.name = val || "Nova";
        state.birthTime = Date.now();
        state.lastUpdated = Date.now();
        updateUI();
        els.overlay.style.opacity = "0";
        setTimeout(() => els.overlay.style.display = "none", 260);
      });

      els.nameInput.addEventListener("keydown", e => { if (e.key === "Enter") els.beginBtn.click(); });
      setTimeout(() => els.nameInput.focus(), 200);

      function computeMood() {
        const { hunger, happiness, energy, love } = state.stats;
        const min = Math.min(hunger, happiness, energy, love);
        const avg = (hunger + happiness + energy + love) / 4;
        if (min === 0) return "critical";
        if (min < 25) return "sad";
        if (avg > 80 && min > 70) return "happy";
        return "normal";
      }

      function setBar(bar, val, pct) {
        const p = Math.round(pct);
        val.textContent = p + "%";
        bar.style.width = p + "%";
        bar.className = "stat-fill " + (p === 0 ? "low" : p <= 25 ? "low" : p <= 65 ? "medium" : "high");
      }

      function updateUI() {
        const { hunger, happiness, energy, love } = state.stats;
        const mood = computeMood();
        const name = state.name || "your friend";

        els.petNameLabel.textContent = state.name || "Unnamed";
        setBar(els.hungerBar, els.hungerValue, hunger);
        setBar(els.happinessBar, els.happinessValue, happiness);
        setBar(els.energyBar, els.energyValue, energy);
        setBar(els.loveBar, els.loveValue, love);

        const faces = { happy: "😊", normal: "✨", sad: "🥺", critical: "😭" };
        const moods = { happy: `${name} is over the moon with joy.`, normal: `${name} feels comfy among the stars.`, sad: `${name} feels a little dim—tend to them.`, critical: `${name} is crying out for you...` };
        const statuses = { happy: "Radiant", normal: "Content", sad: "Fading", critical: "In Tears" };

        els.petFace.textContent = faces[mood];
        els.moodText.textContent = moods[mood];
        els.statusText.textContent = statuses[mood];

        const avg = (hunger + happiness + energy + love) / 4;
        const min = Math.min(hunger, happiness, energy, love);
        els.overallText.textContent = min === 0 ? "Overall: Critical" : min < 25 ? "Overall: Fragile" : avg > 80 ? "Overall: Radiant" : "Overall: Stable";

        els.petWrapper.className = "pet-wrapper mood-" + mood;
        const ageMin = Math.floor((Date.now() - state.birthTime) / 60000);
        els.ageDisplay.textContent = "Age: " + ageMin + " min";
      }

      function adjustStat(key, delta) {
        state.stats[key] = clamp(state.stats[key] + delta, 0, 100);
        updateUI();
      }

      els.btnFeed.addEventListener("click", () => adjustStat("hunger", 18));
      els.btnPlay.addEventListener("click", () => adjustStat("happiness", 16));
      els.btnSleep.addEventListener("click", () => adjustStat("energy", 20));
      els.btnCuddle.addEventListener("click", () => adjustStat("love", 18));

      setInterval(() => {
        const elapsed = (Date.now() - state.lastUpdated) / 1000;
        state.stats.hunger = clamp(state.stats.hunger - 0.06 * elapsed, 0, 100);
        state.stats.happiness = clamp(state.stats.happiness - 0.06 * elapsed, 0, 100);
        state.stats.energy = clamp(state.stats.energy - 0.05 * elapsed, 0, 100);
        state.stats.love = clamp(state.stats.love - 0.05 * elapsed, 0, 100);
        state.lastUpdated = Date.now();
        updateUI();
      }, 5000);

      updateUI();
    })();
  </script>
